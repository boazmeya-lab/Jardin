document.addEventListener('DOMContentLoaded', () => { const grid = document.getElementById('productsGrid'); const filterBtns = document.querySelectorAll('.filter-btn'); // Récupérer la catégorie transmise dans l'URL (ex: ?cat=hotels) const urlParams = new URLSearchParams(window.location.search); const selectedCat = urlParams.get('cat') || 'all'; function displayProducts(items) { if (!grid) return; if (!items || items.length === 0) { grid.innerHTML = '

Aucun produit dans cette catégorie.

'; return; } grid.innerHTML = items.map(p => `
${p.surMesure ? 'Sur mesure' : 'Livrable'}

${p.title}

${p.price}

Découvrir
`).join(''); } // Filtrage initial basé sur l'URL if (typeof products !== 'undefined') { if (selectedCat === 'all') { displayProducts(products); } else { const filtered = products.filter(p => p.category === selectedCat); displayProducts(filtered); // Mettre à jour l'état actif du bouton filterBtns.forEach(b => { if (b.dataset.category === selectedCat) b.classList.add('active'); else b.classList.remove('active'); }); } } // Gestion des clics manuels sur les boutons filterBtns.forEach(btn => { btn.addEventListener('click', () => { filterBtns.forEach(b => b.classList.remove('active')); btn.classList.add('active'); const cat = btn.dataset.category; if (cat === 'all') { displayProducts(products); } else { displayProducts(products.filter(p => p.category === cat)); } }); }); });
